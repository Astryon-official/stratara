# Stratara OS Kernel Command Line
# This file defines the kernel parameters for the Stratara OS ISO

# Default boot (installed system)
DEFAULT quiet splash=silent vt.global_cursor_default=0 loglevel=3 systemd.show_status=auto rd.udev.log_level=3

# Installer boot
INSTALL quiet splash=silent vt.global_cursor_default=0 loglevel=3 systemd.show_status=auto rd.udev.log_level=3 inst.stage2=hd:LABEL=Stratara-OS

# Live boot
LIVE quiet splash=silent vt.global_cursor_default=0 loglevel=3 systemd.show_status=auto rd.udev.log_level=3 rd.live.image

# Fallback (recovery)
RECOVERY single systemd.unit=emergency.target

# Kernel parameters explanation:
# quiet           - Suppress most kernel messages
# splash=silent   - Use silent splash (plymouth)
# vt.global_cursor_default=0 - Hide cursor on VT
# loglevel=3      - Only show errors/warnings
# systemd.show_status=auto - Show status only when needed
# rd.udev.log_level=3 - Reduce udev logging in initrd
# inst.stage2=hd:LABEL=Stratara-OS - Installation stage 2 location
# rd.live.image   - Boot from live image

# Additional parameters for specific hardware:

# NVIDIA: nvidia-drm.modeset=1

# Stratara-specific:
